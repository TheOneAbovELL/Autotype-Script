vector dref = tasks;
int n = dref.size();
int m = shifts.size();
vector pref(n + 1, 0);
for (int i = 0; i < n; i++) {
pref[i + 1] = pref[i] + dref[i];
}
long long total_sum = pref[n];
vector ans(m);
int curr = 0;
long long rem = dref[0];
for (int j = 0; j < m; j++) {
long long t = shifts[j];
if (t >= rem) {
t -= rem;
curr++;
if (curr == n) {
curr = 0;
rem = dref[0];
} else {
rem = dref[curr];
}
if (t >= total_sum) {
t %= total_sum;
}
if (t > 0) {
long long target = pref[curr] + t;
auto it = upper_bound(pref.begin() + curr, pref.end(), target);
int next_idx = distance(pref.begin(), it) - 1;
t -= (pref[next_idx] - pref[curr]);
curr = next_idx;
if (curr == n) {
curr = 0;
rem = dref[0];
} else {
rem = dref[curr] - t;
if (rem == 0) {
curr++;
if (curr == n) {
curr = 0;
rem = dref[0];
} else {
rem = dref[curr];
}
}
}
}
} else {
rem -= t;
}
if (curr == 0 && rem == dref[0]) {
ans[j] = 0;
} else {
ans[j] = n - curr;
}
}
return ans;
